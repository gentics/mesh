1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ . }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "mesh.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "mesh.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "mesh.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.mesh.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "mesh.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:80
{{- end }}

2. Download the API key (using the commercial Gentics Mesh CLI):
  kubectl run {{ template "mesh.fullname" . }}-cli --rm --tty -i --restart='Never' \
    --image {{ .Values.tools.image.repository }}:{{ .Values.tools.image.tag }} \
    --namespace {{ .Release.Namespace }} \
    --command -- sh -c "unset JAVA_TOOL_OPTIONS ; java -jar /mesh/mesh-cli.jar --username admin --password {{ .Values.credentials.initialAdminPassword | default "GENERATED_ADMIN_PASSWORD" }} --url http://{{ include "mesh.fullname" . }}:{{ .Values.service.mesh.port }}/api/v2 user apikey admin"
{{- if not .Values.credentials.initialAdminPassword}}

   (you can find the generated admin password in the logs)
{{- end }}

3. MCP Server ist aktiviert ({{ .Values.mcpServer.transport.mode }}-Modus).
{{- if eq .Values.mcpServer.transport.mode "http" }}

   Service: {{ include "mesh.mcpServer.fullname" . }}.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.mcpServer.transport.http.port }}/mcp

   {{- if .Values.mcpServer.ingress.enabled }}
   Ingress: {{ range .Values.mcpServer.ingress.hosts }}https://{{ .host }} {{ end }}
   {{- end }}

   Bearer-Token abrufen:
   {{- if .Values.mcpServer.transport.auth.disabled }}
     (Auth deaktiviert – Reverse-Proxy übernimmt!)
   {{- else }}
     kubectl get secret {{ include "mesh.mcpServer.authSecretName" . }} \
       -n {{ .Release.Namespace }} -o jsonpath='{.data.MCP_AUTH_TOKEN}' | base64 -d ; echo
   {{- end }}

   Quick-Test:
     kubectl port-forward svc/{{ include "mesh.mcpServer.fullname" . }} {{ .Values.mcpServer.transport.http.port }}:{{ .Values.mcpServer.transport.http.port }} &
     curl -s http://localhost:{{ .Values.mcpServer.transport.http.port }}/mcp \
       -H "Authorization: Bearer <TOKEN>" -H 'Content-Type: application/json' \
       -d '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{}}'
{{- else }}

   stdio-Modus: kein Service, Zugriff via kubectl exec oder Sidecar.
{{- end }}
{{- end }}
